# ----------Functions----------
def standardize_phone(phone):
    if pd.isna(phone):
        return None
    digits = ''.join(filter(str.isdigit, str(phone)))
    return f"+91-{digits[-10:]}" if len(digits) >= 10 else None

def parse_date(value):
    try:
        return parser.parse(str(value)).date()
    except:
        return None
